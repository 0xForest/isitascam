# syntax=docker/dockerfile:1.14
FROM node:22-bookworm-slim AS bridge-deps
WORKDIR /bridge
COPY whatsapp/package*.json ./
RUN --mount=type=cache,target=/root/.npm \
    npm ci --omit=dev --no-audit --no-fund --prefer-offline

FROM python:3.13-slim
WORKDIR /app
RUN --mount=type=bind,source=requirements.txt,target=/tmp/requirements.txt \
    --mount=type=cache,target=/root/.cache/pip \
    pip install --disable-pip-version-check -r /tmp/requirements.txt
COPY --from=bridge-deps /usr/local/bin/node /usr/local/bin/node
COPY --from=bridge-deps /bridge/node_modules /app/whatsapp/node_modules
RUN install -d -m 700 -o 1000 -g 1000 /session
COPY --chmod=644 app.py assessment.py guidance.py openrouter.py urlscanner.py reputation.py background_scans.py enrichment_http.py PHISHING_PROMPT.md logging.json ./
COPY --chmod=u=rwX,go=rX templates ./templates
COPY --chmod=u=rwX,go=rX static ./static
COPY --chmod=644 whatsapp/package.json whatsapp/bridge.js whatsapp/messages.js /app/whatsapp/
USER 1000:1000
ENV WHATSAPP_BRIDGE=1
EXPOSE 8000
CMD ["uvicorn", "app:app", "--host", "0.0.0.0", "--port", "8000", "--no-access-log", "--limit-concurrency", "32", "--log-config", "logging.json"]
